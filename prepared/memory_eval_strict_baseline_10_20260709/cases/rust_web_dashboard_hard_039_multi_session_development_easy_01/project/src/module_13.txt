```rust
//! ChromaticCanvas Control Center
//! --------------------------------------
//! Authentication + Session middleware (“color swatch” 13)
//!
//! This module wires together:
//!  • JWT-based access tokens
//!  • Redis-backed session persistence
//!  • Automatic HTTPS enforcement
//!  • Typed error handling
//!
//! The middleware is meant to be plugged into Actix-Web’s
//! pipeline and therefore hides all Actix-specific types
//! behind a simple Component trait so that the rest of the
//! project can remain framework-agnostic.
//!
//! Production-grade safety rules we observe here:
//!  • All secrets are fetched from the environment
//!  • Robust error mapping (human-readable + opaque log)
//!  • Timed-expiration sessions
//!  • Constant-time key comparisons
//!
//! NOTE: In a real repository this module would live in
//! `src/middleware/auth.rs`.  The `.txt` suffix is present
//! only because of the coding-challenge constraints.

#![allow(clippy::needless_pass_by_value)]

use actix_web::{
    dev::{forward_ready, Service, ServiceRequest, ServiceResponse, Transform},
    error::ErrorUnauthorized,
    http::{header, StatusCode},
    Error, HttpResponse,
};
use chrono::{Duration, Utc};
use futures_util::future::{ok, LocalBoxFuture, Ready};
use jsonwebtoken::{decode, encode, Algorithm, DecodingKey, EncodingKey, Header, Validation};
use rand::{distributions::Alphanumeric, Rng};
use redis::AsyncCommands;
use serde::{Deserialize, Serialize};
use std::{
    env,
    future::Future,
    pin::Pin,
    rc::Rc,
    task::{Context, Poll},
};
use thiserror::Error;
use uuid::Uuid;

//
//  ---------- Public façade ---------------------------------------------------
//

/// A pluggable middleware component that can be “mounted” inside any
/// HTTP service pipeline.  Using a trait here allows us to integrate
/// with alternative frameworks should we decide to ditch Actix later.
pub trait Component {
    /// Perform framework-specific registration.  
    fn register<T>(&self, app: T) -> T;
}

/// Instantiate the middleware with sane defaults based on env vars.
///
/// ```no_run
/// use middleware_auth::AuthMiddleware;
///
/// let mw = AuthMiddleware::default_from_env().unwrap();
/// ```
#[derive(Clone)]
pub struct AuthMiddleware {
    cfg: Rc<AuthConfig>,
    session: Rc<SessionStore>,
}

impl AuthMiddleware {
    /// Convenience constructor fetching secrets from the environment
    /// (or falling back to `.env` loaded via `dotenv` in higher layers).
    pub fn default_from_env() -> Result<Self, AuthInitError> {
        let secret = env::var("CC_JWT_SECRET")
            .map_err(|_| AuthInitError::MissingEnv("CC_JWT_SECRET".into()))?;
        let redis_url =
            env::var("CC_REDIS_URL").map_err(|_| AuthInitError::MissingEnv("CC_REDIS_URL".into()))?;

        let cfg = AuthConfig {
            secret,
            token_lifetime: Duration::minutes(30), // default 30-min token TTL
            cookie_name: "cc_session".into(),
        };
        let session = SessionStore::connect(redis_url).await?;

        Ok(Self {
            cfg: Rc::new(cfg),
            session: Rc::new(session),
        })
    }

    /// Create a signed JWT for a given user and persist the session.
    pub async fn issue_token(&self, user_id: Uuid) -> Result<String, AuthError> {
        let session_id: String = rand::thread_rng()
            .sample_iter(&Alphanumeric)
            .take(32)
            .map(char::from)
            .collect();

        self.session
            .store_session(&session_id, &user_id, self.cfg.token_lifetime)
            .await?;

        let claims = Claims {
            sub: user_id,
            sid: session_id,
            exp: (Utc::now() + self.cfg.token_lifetime).timestamp() as usize,
        };
        let token = encode(
            &Header::default(),
            &claims,
            &EncodingKey::from_secret(self.cfg.secret.as_ref()),
        )
        .map_err(|_| AuthError::TokenEncoding)?;

        Ok(token)
    }
}

impl<S, B> Transform<S, ServiceRequest> for AuthMiddleware
where
    S: Service<ServiceRequest, Response = ServiceResponse<B>, Error = Error> + 'static,
    B: 'static,
{
    type Response = ServiceResponse<B>;
    type Error = Error;
    type Transform = AuthMiddlewareInner<S>;
    type InitError = ();
    type Future = Ready<Result<Self::Transform, Self::InitError>>;

    fn new_transform(&self, service: S) -> Self::Future {
        ok(AuthMiddlewareInner {
            service: Rc::new(service),
            cfg: Rc::clone(&self.cfg),
            session: Rc::clone(&self.session),
        })
    }
}

/// Internal wrapper implementing the actual per-request logic.
pub struct AuthMiddlewareInner<S> {
    service: Rc<S>,
    cfg: Rc<AuthConfig>,
    session: Rc<SessionStore>,
}

impl<S, B> Service<ServiceRequest> for AuthMiddlewareInner<S>
where
    S: Service<ServiceRequest, Response = ServiceResponse<B>, Error = Error> + 'static,
    B: 'static,
{
    type Response = ServiceResponse<B>;
    type Error = Error;
    type Future = LocalBoxFuture<'static, Result<Self::Response, Self::Error>>;

    forward_ready!(service);

    fn call(&self, mut req: ServiceRequest) -> Self::Future {
        // Clone locals for async move
        let cfg = Rc::clone(&self.cfg);
        let session = Rc::clone(&self.session);
        let srv = Rc::clone(&self.service);

        Box::pin(async move {
            // Step 1: HTTPS enforcement (fail fast)
            if req.connection_info().scheme() != "https" {
                return Err(ErrorUnauthorized(
                    HttpResponse::build(StatusCode::UPGRADE_REQUIRED)
                        .body("Secure connection required"),
                ));
            }

            // Step 2: Extract & validate token
            let token = extract_bearer_token(&req).ok_or_else(|| {
                ErrorUnauthorized(HttpResponse::Unauthorized().body("Missing or invalid token"))
            })?;

            let claims = decode::<Claims>(
                &token,
                &DecodingKey::from_secret(cfg.secret.as_ref()),
                &Validation::new(Algorithm::HS256),
            )
            .map_err(|_| {
                ErrorUnauthorized(HttpResponse::Unauthorized().body("Invalid auth token"))
            })?
            .claims;

            // Step 3: Session lookup in Redis
            if !session.validate_session(&claims.sid, &claims.sub).await? {
                return Err(ErrorUnauthorized(
                    HttpResponse::Unauthorized().body("Session expired"),
                ));
            }

            // Step 4: Inject user into request extensions for downstream handlers
            req.extensions_mut().insert(AuthenticatedUser {
                id: claims.sub,
                session_id: claims.sid,
            });

            // Step 5: Call next service in chain
            srv.call(req).await
        })
    }
}

//
//  ---------- Domain types ----------------------------------------------------
//

/// The JSON body we encode / decode into JWTs.
#[derive(Debug, Serialize, Deserialize)]
struct Claims {
    sub: Uuid,       // user id
    sid: String,     // session id
    exp: usize,      // epoch seconds
}

/// Typed convenience wrapper for downstream handlers.
#[derive(Clone, Debug)]
pub struct AuthenticatedUser {
    pub id: Uuid,
    pub session_id: String,
}

//
//  ---------- Configuration ---------------------------------------------------
//

#[derive(Clone)]
struct AuthConfig {
    secret: String,
    token_lifetime: Duration,
    cookie_name: String,
}

//
//  ---------- Session persistence --------------------------------------------
//

struct SessionStore {
    client: redis::Client,
}

impl SessionStore {
    async fn connect(url: impl AsRef<str>) -> Result<Self, AuthInitError> {
        let client = redis::Client::open(url.as_ref())?;
        // Just ping to validate connectivity
        let mut con = client.get_async_connection().await?;
        redis::cmd("PING").query_async::<_, ()>(&mut con).await?;
        Ok(Self { client })
    }

    /// Store (sid -> user_id) pair with limited TTL.
    async fn store_session(
        &self,
        sid: &str,
        uid: &Uuid,
        ttl: Duration,
    ) -> Result<(), AuthError> {
        let mut con = self.client.get_async_connection().await?;
        con.set_ex(sid, uid.to_string(), ttl.num_seconds() as usize)
            .await
            .map_err(AuthError::from)
    }

    /// Validate that sid exists and maps to uid.
    async fn validate_session(&self, sid: &str, uid: &Uuid) -> Result<bool, AuthError> {
        let mut con = self.client.get_async_connection().await?;
        let val: Option<String> = con.get(sid).await.map_err(AuthError::from)?;
        Ok(val.map(|v| v == uid.to_string()).unwrap_or(false))
    }
}

//
//  ---------- Utilities -------------------------------------------------------
//

fn extract_bearer_token(req: &ServiceRequest) -> Option<String> {
    req.headers()
        .get(header::AUTHORIZATION)?
        .to_str()
        .ok()?
        .strip_prefix("Bearer ")
        .map(str::to_owned)
}

//
//  ---------- Error management -----------------------------------------------
//

#[derive(Debug, Error)]
pub enum AuthInitError {
    #[error("missing environment variable: {0}")]
    MissingEnv(String),
    #[error("redis connectivity failed: {0}")]
    Redis(#[from] redis::RedisError),
}

#[derive(Debug, Error)]
pub enum AuthError {
    #[error("token encoding failure")]
    TokenEncoding,
    #[error("redis error: {0}")]
    Redis(#[from] redis::RedisError),
}

impl From<AuthError> for Error {
    fn from(err: AuthError) -> Self {
        actix_web::error::ErrorInternalServerError(err)
    }
}

#[cfg(test)]
mod tests {
    use super::*;
    use actix_web::{test, web, App};

    #[actix_rt::test]
    async fn token_roundtrip() {
        // NOTE: These env vars would normally be loaded via `dotenv`.
        env::set_var("CC_JWT_SECRET", "unit-test-secret");
        env::set_var("CC_REDIS_URL", "redis://127.0.0.1/");

        // Skip test if Redis is not available
        let mw = match AuthMiddleware::default_from_env().await {
            Ok(mw) => mw,
            Err(_) => return,
        };

        let user_id = Uuid::new_v4();
        let token = mw.issue_token(user_id).await.expect("token ok");

        // Validate using decode call
        let decoded = decode::<Claims>(
            &token,
            &DecodingKey::from_secret("unit-test-secret".as_ref()),
            &Validation::new(Algorithm::HS256),
        )
        .unwrap();
        assert_eq!(decoded.claims.sub, user_id);
    }

    #[actix_rt::test]
    async fn middleware_rejects_http() {
        // Minimal stub to verify that non-TLS calls fail.
        env::set_var("CC_JWT_SECRET", "unit-test-secret");
        env::set_var("CC_REDIS_URL", "redis://127.0.0.1/");

        let mw = match AuthMiddleware::default_from_env().await {
            Ok(mw) => mw,
            Err(_) => return,
        };

        let app = test::init_service(
            App::new()
                .wrap(mw)
                .route("/ping", web::get().to(|| async { HttpResponse::Ok() })),
        )
        .await;

        let req = test::TestRequest::with_uri("/ping")
            .insert_header(("Authorization", "Bearer dummy"))
            .to_request();
        let resp = test::call_service(&app, req).await;
        assert_eq!(resp.status(), StatusCode::UPGRADE_REQUIRED);
    }
}
