use serde::{Deserialize, Serialize};

// ------------------------------- Constants -------------------------------------
static TOKEN_TTL: Lazy<Duration> = Lazy::new(|| Duration::hours(12));

// ------------------------------- Error Handling --------------------------------

/// All errors generated by this module bubble up as `AuthError`.
#[derive(Debug)]
pub enum AuthError {
    InvalidToken(String),
    MissingToken,
    Database(anyhow::Error),
    JwtCreation(anyhow::Error),
}

impl fmt::Display for AuthError {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        match self {
            AuthError::InvalidToken(_) => write!(f, "Invalid authentication token"),
            AuthError::MissingToken => write!(f, "Missing authentication token"),
            AuthError::Database(_) => write!(f, "Database error"),
            AuthError::JwtCreation(_) => write!(f, "Token creation error"),
        }
    }
}

impl ResponseError for AuthError {
    fn status_code(&self) -> StatusCode {
        match self {
            AuthError::MissingToken | AuthError::InvalidToken(_) => StatusCode::UNAUTHORIZED,
            AuthError::Database(_) | AuthError::JwtCreation(_) => StatusCode::INTERNAL_SERVER_ERROR,
        }
    }

    fn error_response(&self) -> HttpResponse<EitherBody<String>> {
        HttpResponse::build(self.status_code()).body(self.to_string())
    }
}

// ------------------------------- JWT Claims ------------------------------------

#[derive(Debug, Serialize, Deserialize)]
pub struct SessionClaims {
    /// Unique identifier of the JWT itself, used for revocation.
    pub jti: Uuid,
    /// Subject (user id) this token grants access for.
    pub sub: Uuid,
    /// Issued at (Unix timestamp seconds)
    pub iat: i64,
    /// Expiration (Unix timestamp seconds)
    pub exp: i64,
}

/// Helper to create a default header (HS256) w/ typ = JWT
fn default_header() -> Header {
    Header {
        alg: Algorithm::HS256,
        ..Default::default()
    }
}

// ------------------------------- Secret Store ----------------------------------

/// Thread-safe container for rotating secrets.
///
/// A production system would hydrate this via KMS or Vault.  For the sake
/// of example we allow an arbitrary number of “active” secrets.  The first
/// entry is treated as the writer; all entries are valid for decoding.
#[derive(Clone)]
pub struct SecretStore {
    inner: Arc<Vec<String>>,
}

impl SecretStore {
    /// Parse a colon-separated list of secrets (`"key1:key2:oldkey3"`)
    pub fn new(raw: &str) -> Self {
        let keys: Vec<String> = raw.split(':').map(|s| s.trim().to_owned()).collect();
        SecretStore {
            inner: Arc::new(keys),
        }
    }

    fn current_key(&self) -> &str {
        &self.inner[0]
    }

    fn encoding_key(&self) -> EncodingKey {
        EncodingKey::from_secret(self.current_key().as_bytes())
    }

    fn decoding_keys(&self) -> Vec<DecodingKey> {
        self.inner
            .iter()
            .map(|k| DecodingKey::from_secret(k.as_bytes()))
            .collect()
    }
}

// ------------------------------- Session Persistence ---------------------------

/// Stores the session in DB for token revocation, analytics, etc.
pub async fn persist_session(
    pool: &PgPool,
    claims: &SessionClaims,
) -> Result<(), AuthError> {
    sqlx::query(
        r#"
        INSERT INTO sessions (jti, user_id, issued_at, expires_at)
        VALUES ($1, $2, to_timestamp($3), to_timestamp($4))
        "#,
    )
    .bind(claims.jti)
    .bind(claims.sub)
    .bind(claims.iat)
    .bind(claims.exp)
    .execute(pool)
    .await
    .map(|_| ())
    .map_err(|e| AuthError::Database(e.into()))
}

/// Validate the token has not been revoked / expired by checking DB.
pub async fn validate_session_in_db(pool: &PgPool, jti: Uuid) -> Result<(), AuthError> {
    let row: Option<(bool,)> = sqlx::query_as("SELECT revoked FROM sessions WHERE jti = $1")
        .bind(jti)
        .fetch_optional(pool)
        .await
        .map_err(|e| AuthError::Database(e.into()))?;

    match row {
        Some((revoked,)) if revoked => Err(AuthError::InvalidToken("revoked".into())),
        Some(_) => Ok(()),
        None => Err(AuthError::InvalidToken("unknown_jti".into())),
    }
}

// ------------------------------- Token API -------------------------------------

/// Issue a fresh JWT for a given `user_id`.
pub async fn issue_token(
    pool: &PgPool,
    secrets: &SecretStore,
    user_id: Uuid,
) -> Result<String, AuthError> {
    let now = OffsetDateTime::now_utc();
    let claims = SessionClaims {
        jti: Uuid::new_v4(),
        sub: user_id,
        iat: now.unix_timestamp(),
        exp: (now + *TOKEN_TTL).unix_timestamp(),
    };

    let token = encode(&default_header(), &claims, &secrets.encoding_key())
        .map_err(|e| AuthError::JwtCreation(e.into()))?;

    persist_session(pool, &claims).await?;
    Ok(token)
}

/// Decode, verify, and validate the token against secrets & database.
pub async fn decode_and_validate(
    pool: &PgPool,
    secrets: &SecretStore,
    token: &str,
) -> Result<SessionClaims, AuthError> {
    // Attempt decoding with any active key.
    let mut last_err = None;
    for key in secrets.decoding_keys() {
        match decode::<SessionClaims>(
            token,
            &key,
            &Validation::new(Algorithm::HS256),
        ) {
            Ok(data) => {
                // Additional DB validation
                validate_session_in_db(pool, data.claims.jti).await?;
                return Ok(data.claims);
            }
            Err(e) => last_err = Some(e),
        }
    }

    Err(AuthError::InvalidToken(
        last_err.map(|e| e.to_string()).unwrap_or_default(),
    ))
}

// ------------------------------- Extractor Utilities ---------------------------

/// Return `Some(token_str)` if the request has a usable Bearer token.
fn extract_token(req: &ServiceRequest) -> Option<String> {
    req.headers()
        .get(header::AUTHORIZATION)
        .and_then(|value| value.to_str().ok())
        .and_then(|hdr| hdr.strip_prefix("Bearer "))
        .map(|s| s.trim().to_owned())
}

// ------------------------------- Middleware Implementation --------------------

/// Key under which the verified user id is added to request extensions.
pub struct AuthenticatedUserId;

impl actix_web::FromRequest for AuthenticatedUserId {
    type Error = AuthError;
    type Future = Ready<Result<Self, AuthError>>;
    type Config = ();

    fn from_request(
        req: &actix_web::HttpRequest,
        _payload: &mut actix_web::dev::Payload,
    ) -> Self::Future {
        match req.extensions().get::<Uuid>() {
            Some(_) => ready(Ok(AuthenticatedUserId)),
            None => ready(Err(AuthError::MissingToken)),
        }
    }
}

/// Main middleware struct implementing `Transform`.
#[derive(Default, Clone)]
pub struct AuthMiddleware;

impl<S, B> Transform<S, ServiceRequest> for AuthMiddleware
where
    S: Service<ServiceRequest, Response = ServiceResponse<B>, Error = actix_web::Error>
        + 'static,
    B: 'static,
{
    type Response = ServiceResponse<EitherBody<B>>;
    type Error = actix_web::Error;
    type InitError = ();
    type Transform = AuthMiddlewareService<S>;
    type Future = StdReady<Result<Self::Transform, Self::InitError>>;

    fn new_transform(&self, service: S) -> Self::Future {
        ready(Ok(AuthMiddlewareService {
            service: Rc::new(service),
        }))
    }
}

pub struct AuthMiddlewareService<S> {
    service: Rc<S>,
}

impl<S, B> Service<ServiceRequest> for AuthMiddlewareService<S>
where
    S: Service<ServiceRequest, Response = ServiceResponse<B>, Error = actix_web::Error>
        + 'static,
    B: 'static,
{
    type Response = ServiceResponse<EitherBody<B>>;
    type Error = actix_web::Error;
    type Future = LocalBoxFuture<'static, Result<Self::Response, Self::Error>>;

    forward_ready!(service);

    fn call(&self, mut req: ServiceRequest) -> Self::Future {
        let svc = Rc::clone(&self.service);

        // Clone needed data for async move
        let token_opt = extract_token(&req);
        let pool_opt = req.app_data::<PgPool>().cloned();
        let secret_store_opt = req.app_data::<SecretStore>().cloned();

        Box::pin(async move {
            // Early exit when no auth header; treat as 401.
            let token = match token_opt {
                Some(t) => t,
                None => {
                    return Err(AuthError::MissingToken.into());
                }
            };

            let pool = pool_opt.ok_or_else(|| AuthError::Database(anyhow::anyhow!("No pool")))?;
            let secrets =
                secret_store_opt.ok_or_else(|| AuthError::InvalidToken("missing secrets".into()))?;

            // Validate token
            let claims = decode_and_validate(&pool, &secrets, &token).await.map_err(|e| {
                let http_err: actix_web::Error = e.into();
                http_err
            })?;

            // Insert user id into extensions for downstream handlers.
            req.extensions_mut().insert(claims.sub);

            // Call downstream service
            let res = svc.call(req).await?.map_into_left_body();
            Ok(res)
        })
    }
}
