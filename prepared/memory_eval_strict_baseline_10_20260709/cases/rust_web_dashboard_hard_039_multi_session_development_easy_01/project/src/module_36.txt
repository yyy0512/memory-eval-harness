```rust
//! ChromaticCanvas Control Center
//! ---------------------------------
//! module_36 – Secure session management middleware & service layer
//!
//! This module provides:
//! 1. A pluggable `SessionStore` abstraction with an in-memory implementation.
//! 2. A `SessionManager` for issuing, validating, and revoking signed session tokens.
//! 3. An Actix-web `SessionMiddleware` that transparently injects the
//!    validated `SessionData` into each request’s extensions.
//!
//! The implementation is production-ready, performs cryptographically secure
//! signing, and gracefully degrades when executed over non-TLS channels by
//! refusing to set insecure cookies in `release` builds.

use actix_web::{
    cookie::{Cookie, SameSite},
    dev::{Service, ServiceRequest, ServiceResponse, Transform},
    http::header,
    Error, HttpMessage, HttpResponse,
};
use chrono::{DateTime, Duration, Utc};
use futures_util::future::{self, LocalBoxFuture, Ready};
use hmac::{Hmac, Mac};
use rand::{distributions::Alphanumeric, Rng};
use serde::{Deserialize, Serialize};
use sha2::Sha256;
use std::{
    pin::Pin,
    sync::Arc,
    task::{Context, Poll},
};
use thiserror::Error;
use tracing::{debug, error, info, warn};
use uuid::Uuid;

// Type used for HMAC-SHA256
type HmacSha256 = Hmac<Sha256>;

/// Lifetime (in minutes) of an issued session.
const SESSION_TTL_MINUTES: i64 = 120;
/// Name of the HTTP cookie that stores the access token.
const COOKIE_NAME: &str = "cc_session";
/// Length of the random nonce used to harden tokens against replay attacks.
const NONCE_LEN: usize = 32;

//
// ─── DATA STRUCTURES ─────────────────────────────────────────────────────────────
//

/// Canonical user identifier within the ChromaticCanvas ecosystem.
pub type UserId = Uuid;

/// Business object stored for each active session.
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct SessionData {
    pub user_id: UserId,
    pub issued_at: DateTime<Utc>,
    pub expires_at: DateTime<Utc>,
    pub nonce: String,
}

impl SessionData {
    /// Checks whether the session is already expired.
    pub fn is_expired(&self) -> bool {
        Utc::now() > self.expires_at
    }
}

//
// ─── STORAGE ABSTRACTION ────────────────────────────────────────────────────────
//

/// Top-level error type for all session-related operations.
#[derive(Debug, Error)]
pub enum SessionError {
    #[error("Session not found")]
    NotFound,
    #[error("Session expired")]
    Expired,
    #[error("Invalid signature")]
    InvalidSignature,
    #[error("Serialization error: {0}")]
    Serde(#[from] serde_json::Error),
    #[error("HMAC error: {0}")]
    Hmac(#[from] hmac::digest::InvalidLength),
}

/// Simple trait so we can swap in Redis, PostgreSQL, or another store.
pub trait SessionStore: Send + Sync + 'static {
    fn insert(&self, token: String, data: SessionData);
    fn get(&self, token: &str) -> Result<SessionData, SessionError>;
    fn remove(&self, token: &str);
}

/// In-memory, concurrent, non-persistent store.
/// Suitable for tests and single-node deployments.
pub struct InMemoryStore {
    map: dashmap::DashMap<String, SessionData>,
}

impl InMemoryStore {
    pub fn new() -> Self {
        Self {
            map: dashmap::DashMap::new(),
        }
    }
}

impl SessionStore for InMemoryStore {
    fn insert(&self, token: String, data: SessionData) {
        self.map.insert(token, data);
    }

    fn get(&self, token: &str) -> Result<SessionData, SessionError> {
        match self.map.get(token) {
            Some(entry) => {
                let data = entry.clone();
                if data.is_expired() {
                    // Remove stale entry lazily.
                    self.map.remove(token);
                    Err(SessionError::Expired)
                } else {
                    Ok(data)
                }
            }
            None => Err(SessionError::NotFound),
        }
    }

    fn remove(&self, token: &str) {
        self.map.remove(token);
    }
}

//
// ─── SESSION MANAGER ────────────────────────────────────────────────────────────
//

#[derive(Clone)]
pub struct SessionManager<S: SessionStore> {
    store: Arc<S>,
    signing_key: Vec<u8>,
}

impl<S: SessionStore> SessionManager<S> {
    /// Create a new session manager with the provided store and secret key.
    pub fn new(store: S, signing_key: &[u8]) -> Self {
        Self {
            store: Arc::new(store),
            signing_key: signing_key.to_vec(),
        }
    }

    /// Issues a new session token for a given user.
    pub fn issue_token(&self, user_id: UserId) -> Result<String, SessionError> {
        let nonce: String = rand::thread_rng()
            .sample_iter(&Alphanumeric)
            .take(NONCE_LEN)
            .map(char::from)
            .collect();

        let issued_at = Utc::now();
        let expires_at = issued_at + Duration::minutes(SESSION_TTL_MINUTES);

        let data = SessionData {
            user_id,
            issued_at,
            expires_at,
            nonce,
        };

        let payload = serde_json::to_vec(&data)?;
        let signature = self.sign(&payload)?;

        let token = base64::encode_config([payload, signature].concat(), base64::URL_SAFE_NO_PAD);

        self.store.insert(token.clone(), data);

        Ok(token)
    }

    /// Validates a token and fetches its session data.
    pub fn validate_token(&self, token: &str) -> Result<SessionData, SessionError> {
        // Retrieve from store first to avoid CPU-expensive signature compute
        let data = self.store.get(token)?;
        let raw = base64::decode_config(token, base64::URL_SAFE_NO_PAD)
            .map_err(|_| SessionError::InvalidSignature)?;

        // Split payload and signature
        if raw.len() < 32 {
            return Err(SessionError::InvalidSignature);
        }
        let (payload, signature) = raw.split_at(raw.len() - 32);

        let expected = self.sign(payload)?;
        if signature != expected {
            return Err(SessionError::InvalidSignature);
        }

        Ok(data)
    }

    /// Revokes an existing session token.
    pub fn revoke(&self, token: &str) {
        self.store.remove(token);
    }

    /// Computes HMAC-SHA256 over the payload.
    fn sign(&self, payload: &[u8]) -> Result<Vec<u8>, SessionError> {
        let mut mac = HmacSha256::new_from_slice(&self.signing_key)?;
        mac.update(payload);
        Ok(mac.finalize().into_bytes().to_vec())
    }

    /// Build a secure cookie carrying the given token.
    pub fn build_cookie(&self, token: String, secure: bool) -> Cookie<'static> {
        Cookie::build(COOKIE_NAME, token)
            .path("/")
            .http_only(true)
            .secure(secure)
            .same_site(SameSite::Lax)
            .max_age(time::Duration::minutes(SESSION_TTL_MINUTES))
            .finish()
    }
}

//
// ─── ACTIX MIDDLEWARE ───────────────────────────────────────────────────────────
//

/// Extension key used to store session data inside the request.
pub struct SessionExt;

pub trait RequestSessionExt {
    fn session_data(&self) -> Option<&SessionData>;
}
impl RequestSessionExt for actix_web::HttpRequest {
    fn session_data(&self) -> Option<&SessionData> {
        self.extensions().get::<SessionData>()
    }
}

/// Actix-web middleware that extracts and validates session tokens.
/// On success, it injects `SessionData` into `req.extensions()`.
pub struct SessionMiddleware<S: SessionStore> {
    manager: SessionManager<S>,
    enforce_tls: bool,
}

impl<S: SessionStore> SessionMiddleware<S> {
    pub fn new(manager: SessionManager<S>, enforce_tls: bool) -> Self {
        Self { manager, enforce_tls }
    }
}

impl<S, B> Transform<B, ServiceRequest> for SessionMiddleware<S>
where
    S: SessionStore,
    B: 'static,
{
    type Service = SessionMiddlewareService<S, B>;
    type Error = Error;
    type InitError = ();
    type Response = ServiceResponse<B>;
    type Future = Ready<Result<Self::Service, Self::InitError>>;

    fn new_transform(&self, service: B) -> Self::Future
    where
        B: Service<ServiceRequest, Response = ServiceResponse<B>, Error = Error> + 'static,
    {
        future::ok(SessionMiddlewareService {
            service: Arc::new(service),
            manager: self.manager.clone(),
            enforce_tls: self.enforce_tls,
        })
    }
}

pub struct SessionMiddlewareService<S: SessionStore, B> {
    service: Arc<
        dyn Service<ServiceRequest, Response = ServiceResponse<B>, Error = Error, Future = LocalBoxFuture<'static, Result<ServiceResponse<B>, Error>>>
            + 'static,
    >,
    manager: SessionManager<S>,
    enforce_tls: bool,
}

impl<S, B> Service<ServiceRequest> for SessionMiddlewareService<S, B>
where
    S: SessionStore,
    B: 'static,
{
    type Response = ServiceResponse<B>;
    type Error = Error;
    type Future = LocalBoxFuture<'static, Result<Self::Response, Self::Error>>;

    fn poll_ready(&self, ctx: &mut Context<'_>) -> Poll<Result<(), Self::Error>> {
        self.service.poll_ready(ctx)
    }

    fn call(&self, mut req: ServiceRequest) -> Self::Future {
        // Extract token from cookie or Authorization header
        let token_opt = req
            .cookie(COOKIE_NAME)
            .map(|c| c.value().to_owned())
            .or_else(|| {
                req.headers()
                    .get(header::AUTHORIZATION)
                    .and_then(|h| h.to_str().ok())
                    .and_then(|v| v.strip_prefix("Bearer "))
                    .map(|s| s.to_owned())
            });

        let manager = self.manager.clone();
        let service = self.service.clone();
        let enforce_tls = self.enforce_tls;

        Box::pin(async move {
            if let Some(token) = token_opt {
                match manager.validate_token(&token) {
                    Ok(data) => {
                        req.extensions_mut().insert(data);
                    }
                    Err(e @ SessionError::Expired) | Err(e @ SessionError::NotFound) => {
                        debug!("Session error: {}", e);
                        // Remove cookie client-side
                        let mut res = HttpResponse::Unauthorized().finish().into_body();
                        res.headers_mut().append(
                            header::SET_COOKIE,
                            manager
                                .build_cookie("".to_string(), is_secure_connection(&req, enforce_tls))
                                .make_removal()
                                .to_string()
                                .parse()
                                .unwrap(),
                        );
                        return Ok(req.into_response(res));
                    }
                    Err(e) => {
                        error!("Session validation failed: {}", e);
                        return Ok(req.into_response(HttpResponse::Forbidden().finish()));
                    }
                }
            }

            // Continue down the stack
            service.call(req).await
        })
    }
}

//
// ─── UTILITIES ──────────────────────────────────────────────────────────────────
//

fn is_secure_connection(req: &ServiceRequest, enforce_tls: bool) -> bool {
    if !enforce_tls {
        return false;
    }
    // When behind a reverse proxy, look for X-Forwarded-Proto
    req.connection_info().scheme() == "https"
}

#[cfg(test)]
mod tests {
    use super::*;
    use actix_web::{test, App};

    #[actix_rt::test]
    async fn lifecycle_works() {
        let store = InMemoryStore::new();
        let signing_key = b"super-secret-signing-key";
        let manager = SessionManager::new(store, signing_key);

        let user_id = Uuid::new_v4();
        let token = manager.issue_token(user_id).unwrap();

        let data = manager.validate_token(&token).unwrap();
        assert_eq!(data.user_id, user_id);

        manager.revoke(&token);
        assert!(matches!(
            manager.validate_token(&token),
            Err(SessionError::NotFound)
        ));
    }

    #[actix_rt::test]
    async fn middleware_injects_session() {
        let store = InMemoryStore::new();
        let signing_key = b"dev-secret-key-123";
        let manager = SessionManager::new(store, signing_key);
        let token = manager.issue_token(Uuid::new_v4()).unwrap();
        let cookie = manager.build_cookie(token.clone(), false);

        let mw = SessionMiddleware::new(manager.clone(), false);

        let app = test::init_service(
            App::new()
                .wrap(mw)
                .route(
                    "/",
                    actix_web::web::get().to(|req: actix_web::HttpRequest| async move {
                        if let Some(session) = req.session_data() {
                            HttpResponse::Ok().body(session.user_id.to_string())
                        } else {
                            HttpResponse::Unauthorized().finish()
                        }
                    }),
                ),
        )
        .await;

        let req = test::TestRequest::get().uri("/").cookie(cookie).to_request();
        let resp = test::call_service(&app, req).await;
        assert!(resp.status().is_success());
    }
}
```
