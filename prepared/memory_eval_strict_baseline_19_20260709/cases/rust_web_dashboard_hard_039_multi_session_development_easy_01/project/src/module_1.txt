```rust
//! ChromaticCanvas Control Center
//! -----------------------------------
//! module_1: Authentication & Session middleware
//!
//! This module provides a production-grade authentication layer built on top
//! of Actix-Web’s `Transform` trait.  It performs the following duties:
//!   • Extract a JWT (`Authorization: Bearer <token>`) from each request
//!   • Validate & decode the token, turning it into strongly-typed `Claims`
//!   • Hydrate a `Session` object from an external store (e.g. Redis)
//!   • Attach the session to the request’s `extensions` so downstream
//!     handlers and components can retrieve it cheaply
//!   • Surface rich, structured errors for fine-grained observability
//!
//! The middleware is intentionally generic over its underlying `SessionStore`
//! implementation to enable dependency injection in unit/integration tests.
//!
//! # Example
//!
//! ```no_run
//! use actix_web::{web, App, HttpServer};
//! use chromatic_canvas::auth::{AuthSessionMiddleware, InMemoryStore};
//!
//! #[actix_web::main]
//! async fn main() -> std::io::Result<()> {
//!     HttpServer::new(|| {
//!         App::new()
//!             .wrap(
//!                 AuthSessionMiddleware::new(
//!                     InMemoryStore::default(),
//!                     "super-secret-signing-key".as_bytes(),
//!                 )
//!             )
//!             .service(
//!                 web::resource("/me").to(|session: chromatic_canvas::auth::Session| async move {
//!                     format!("Hello, {}!", session.user_email)
//!                 })
//!             )
//!     })
//!     .bind("0.0.0.0:8080")?
//!     .run()
//!     .await
//! }
//! ```
// -------------------------------------------------------------------------------------------------

use actix_web::dev::{forward_ready, Service, ServiceRequest, ServiceResponse, Transform};
use actix_web::http::header::AUTHORIZATION;
use actix_web::{Error as ActixError, HttpMessage, HttpResponse};
use futures_util::future::{LocalBoxFuture, Ready};
use jsonwebtoken::{decode, Algorithm, DecodingKey, Validation};
use log::{error, info, warn};
use serde::{Deserialize, Serialize};
use std::future::ready;
use std::rc::Rc;
use std::task::{Context, Poll};
use std::time::{Duration, SystemTime, UNIX_EPOCH};
use thiserror::Error;
use uuid::Uuid;

// -------------------------------------------------------------------------------------------------
// Domain models
// -------------------------------------------------------------------------------------------------

/// Strongly-typed representation of the JWT we issue to browsers.
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct Claims {
    /// Unique, stable identifier for the ChromaticCanvas user account
    pub sub: Uuid,
    /// ISO-8601 email address
    pub email: String,
    /// When the token was issued (seconds since epoch)
    pub iat: u64,
    /// Expiration timestamp (seconds since epoch)
    pub exp: u64,
}

/// Session object that lives for the duration of a single HTTP request.
///
/// Unlike `Claims`, the `Session` may be hydrated with additional information
/// such as roles or feature flags (not shown here to keep the example focused).
#[derive(Debug, Clone)]
pub struct Session {
    pub user_id: Uuid,
    pub user_email: String,
    pub issued_at: SystemTime,
    pub expires_at: SystemTime,
}

// -------------------------------------------------------------------------------------------------
// Error definitions
// -------------------------------------------------------------------------------------------------

/// Enumeration of all authentication/session-related failures.
#[derive(Debug, Error)]
pub enum AuthError {
    #[error("missing `Authorization` header")]
    MissingAuthHeader,

    #[error("malformed `Authorization` header")]
    MalformedAuthHeader,

    #[error("invalid or expired token")]
    InvalidToken,

    #[error("session not found")]
    SessionNotFound,

    #[error("internal error: {0}")]
    Internal(#[from] anyhow::Error),
}

impl From<AuthError> for ActixError {
    fn from(err: AuthError) -> Self {
        let status = match &err {
            AuthError::MissingAuthHeader
            | AuthError::MalformedAuthHeader
            | AuthError::InvalidToken
            | AuthError::SessionNotFound => actix_web::http::StatusCode::UNAUTHORIZED,
            AuthError::Internal(_) => actix_web::http::StatusCode::INTERNAL_SERVER_ERROR,
        };

        HttpResponse::build(status).json(serde_json::json!({
            "error": err.to_string(),
            "code": status.as_u16(),
        }))
        .into()
    }
}

// -------------------------------------------------------------------------------------------------
// Session store abstraction
// -------------------------------------------------------------------------------------------------

/// Persisted state for a given user across requests.
#[derive(Debug, Serialize, Deserialize, Clone)]
pub struct PersistedSession {
    pub user_id: Uuid,
    pub user_email: String,
    pub expires_at: SystemTime,
}

/// Abstraction over any key-value store that can retrieve a [`PersistedSession`].
///
/// Implementations may choose Redis, DynamoDB, in-memory hash map, etc.
pub trait SessionStore: Clone + 'static {
    /// Fetch the session by user-id. Returns `None` if the session does not exist
    /// or has expired.
    fn fetch(&self, user_id: Uuid) -> Result<Option<PersistedSession>, anyhow::Error>;
}

/// A bare-bones in-memory store useful for tests and local development.
#[derive(Default, Clone)]
pub struct InMemoryStore {
    inner: Rc<std::sync::Mutex<std::collections::HashMap<Uuid, PersistedSession>>>,
}

impl SessionStore for InMemoryStore {
    fn fetch(&self, user_id: Uuid) -> Result<Option<PersistedSession>, anyhow::Error> {
        Ok(self.inner.lock().unwrap().get(&user_id).cloned())
    }
}

// -------------------------------------------------------------------------------------------------
// Middleware definition
// -------------------------------------------------------------------------------------------------

/// Authentication + session resolution middleware.
pub struct AuthSessionMiddleware<S, Store> {
    service: Rc<S>,
    session_store: Store,
    key: Rc<DecodingKey>,
}

impl<S, Store> AuthSessionMiddleware<S, Store> {
    pub fn new(service: S, session_store: Store, jwt_secret: &[u8]) -> Self {
        Self {
            service: Rc::new(service),
            session_store,
            key: Rc::new(DecodingKey::from_secret(jwt_secret)),
        }
    }
}

impl<S, Store, B> Transform<S, ServiceRequest> for AuthSessionMiddleware<S, Store>
where
    S: Service<ServiceRequest, Response = ServiceResponse<B>, Error = ActixError> + 'static,
    S::Future: 'static,
    Store: SessionStore,
    B: 'static,
{
    type Response = ServiceResponse<B>;
    type Error = ActixError;
    type Transform = AuthSessionMiddlewareService<S, Store>;
    type InitError = ();
    type Future = Ready<Result<Self::Transform, Self::InitError>>;

    fn new_transform(&self, service: S) -> Self::Future {
        ready(Ok(AuthSessionMiddlewareService {
            service: Rc::new(service),
            session_store: self.session_store.clone(),
            key: self.key.clone(),
        }))
    }
}

/// The actual service that Actix calls per request.
pub struct AuthSessionMiddlewareService<S, Store> {
    service: Rc<S>,
    session_store: Store,
    key: Rc<DecodingKey>,
}

impl<S, Store, B> Service<ServiceRequest> for AuthSessionMiddlewareService<S, Store>
where
    S: Service<ServiceRequest, Response = ServiceResponse<B>, Error = ActixError> + 'static,
    S::Future: 'static,
    Store: SessionStore,
{
    type Response = ServiceResponse<B>;
    type Error = ActixError;
    type Future = LocalBoxFuture<'static, Result<Self::Response, Self::Error>>;

    forward_ready!(service);

    fn call(&self, mut req: ServiceRequest) -> Self::Future {
        let key = self.key.clone();
        let store = self.session_store.clone();
        let srv = self.service.clone();

        Box::pin(async move {
            // Extract & validate the token --------------------------------------------------------
            let token = extract_token(&req).map_err(AuthError::from)?;
            let claims = validate_token(&token, &key).map_err(AuthError::from)?;

            // Hydrate the session ----------------------------------------------------------------
            let persisted = store
                .fetch(claims.sub)
                .map_err(|e| AuthError::Internal(e.into()))?
                .ok_or(AuthError::SessionNotFound)?;

            // Convert to runtime `Session`
            let session = Session {
                user_id: persisted.user_id,
                user_email: persisted.user_email,
                issued_at: unix_epoch(claims.iat),
                expires_at: persisted.expires_at,
            };

            // Attach session to request extensions
            req.extensions_mut().insert(session.clone());

            // Logging helpers
            info!(
                "`{}` authenticated (expires {:?})",
                session.user_email,
                session.expires_at
            );

            // Continue down the service chain
            let res = srv.call(req).await?;

            Ok(res)
        })
    }
}

// -------------------------------------------------------------------------------------------------
// Helper functions
// -------------------------------------------------------------------------------------------------

fn extract_token(req: &ServiceRequest) -> Result<String, AuthError> {
    let header = req
        .headers()
        .get(AUTHORIZATION)
        .ok_or(AuthError::MissingAuthHeader)?
        .to_str()
        .map_err(|_| AuthError::MalformedAuthHeader)?;

    if !header.starts_with("Bearer ") {
        return Err(AuthError::MalformedAuthHeader);
    }

    Ok(header.trim_start_matches("Bearer ").to_owned())
}

fn validate_token(token: &str, key: &DecodingKey) -> Result<Claims, AuthError> {
    let validation = Validation::new(Algorithm::HS256);
    let data = decode::<Claims>(token, key, &validation).map_err(|e| {
        warn!("JWT validation failure: {}", e);
        AuthError::InvalidToken
    })?;
    Ok(data.claims)
}

fn unix_epoch(secs: u64) -> SystemTime {
    UNIX_EPOCH + Duration::from_secs(secs)
}

// -------------------------------------------------------------------------------------------------
// Extension extractor
// -------------------------------------------------------------------------------------------------

use actix_web::{
    dev::Payload,
    FromRequest,
    HttpRequest,
};
use futures_util::future::ready as fut_ready;

/// Seamlessly pull the `Session` out of request extensions inside handlers.
///
/// Example:
/// ```ignore
/// async fn me(session: Session) -> impl Responder {
///     HttpResponse::Ok().json(session)
/// }
/// ```
impl FromRequest for Session {
    type Error = ActixError;
    type Future = Ready<Result<Self, Self::Error>>;
    type Config = ();

    fn from_request(req: &HttpRequest, _payload: &mut Payload) -> Self::Future {
        if let Some(sess) = req.extensions().get::<Session>().cloned() {
            fut_ready(Ok(sess))
        } else {
            fut_ready(Err(AuthError::SessionNotFound.into()))
        }
    }
}

// Tests (run with `cargo test --package chromatic_canvas -- auth_middleware`)

#[cfg(test)]
mod tests {
    use super::*;
    use actix_service::Service as _; // for call_service
    use actix_web::{test, web, App, HttpResponse};

    fn make_token(sub: Uuid, email: &str, key: &[u8]) -> String {
        let claims = Claims {
            sub,
            email: email.into(),
            iat: 0,
            exp: (SystemTime::now()
                .duration_since(UNIX_EPOCH)
                .unwrap_or_default()
                + Duration::from_secs(60))
            .as_secs(),
        };

        jsonwebtoken::encode(
            &jsonwebtoken::Header::default(),
            &claims,
            &jsonwebtoken::EncodingKey::from_secret(key),
        )
        .unwrap()
    }

    #[actix_rt::test]
    async fn rejects_missing_header() {
        let store = InMemoryStore::default();
        let key = b"secret";

        let app = test::init_service(
            App::new()
                .wrap(AuthSessionMiddleware::new(
                    test::fn_service(|req| async { Ok(req.into_response(HttpResponse::Ok())) }),
                    store,
                    key,
                ))
                .service(web::resource("/").to(|| async { HttpResponse::Ok() })),
        )
        .await;

        let req = test::TestRequest::with_uri("/").to_request();
        let res = test::call_service(&app, req).await;
        assert_eq!(res.status(), actix_web::http::StatusCode::UNAUTHORIZED);
    }

    #[actix_rt::test]
    async fn accepts_valid_token() {
        let user_id = Uuid::new_v4();
        let email = "artist@example.com";
        let key = b"secret";

        let persisted = PersistedSession {
            user_id,
            user_email: email.into(),
            expires_at: SystemTime::now() + Duration::from_secs(60),
        };

        let mut store = InMemoryStore::default();
        store
            .inner
            .lock()
            .unwrap()
            .insert(user_id, persisted.clone());

        let token = make_token(user_id, email, key);

        let app = test::init_service(
            App::new()
                .wrap(AuthSessionMiddleware::new(
                    test::fn_service(|req| async { Ok(req.into_response(HttpResponse::Ok())) }),
                    store.clone(),
                    key,
                ))
                .service(web::resource("/").to(|session: Session| async move {
                    HttpResponse::Ok().body(format!("Hello {}", session.user_email))
                })),
        )
        .await;

        let req = test::TestRequest::with_uri("/")
            .insert_header((AUTHORIZATION, format!("Bearer {}", token)))
            .to_request();

        let res = test::call_service(&app, req).await;
        assert_eq!(res.status(), actix_web::http::StatusCode::OK);
    }
}
