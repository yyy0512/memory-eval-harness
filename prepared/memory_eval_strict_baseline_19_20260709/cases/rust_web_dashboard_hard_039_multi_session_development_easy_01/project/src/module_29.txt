```rust
//! module_29 – Secure session handling & middleware for ChromaticCanvas Control Center
//!
//! This module encapsulates stateless‒but-revocable JSON-Web-Token sessions
//! and provides an Actix-compatible extractor so every REST endpoint can
//! easily gain authenticated-user context.
//!
//! Core ideas
//! ----------
//! • JWTs are signed with HS256 and *also* stored server–side so we can
//!   invalidate them at logout or after privilege escalation.
//! • A thin, concurrent `DashMap` acts as the authoritative session store
//!   (backed by Redis or Postgres in production through the same trait).
//! • The extractor (`Session`) plugs into Actix (`FromRequest`) and turns
//!   common auth failures into typed `SessionError`s so UI code can map them
//!   to toast messages, redirects, etc.

use std::{
    future::Future,
    pin::Pin,
    sync::Arc,
    task::{Context, Poll},
    time::Duration as StdDuration,
};

use actix_web::{
    dev::{forward_ready, Service, ServiceRequest, ServiceResponse, Transform},
    error::ResponseError,
    http::{header, StatusCode},
    FromRequest, HttpMessage, HttpRequest, Result,
};
use chrono::{DateTime, Duration, Utc};
use dashmap::DashMap;
use futures_util::future::{local_boxed, LocalBoxFuture, Ready};
use jsonwebtoken::{decode, encode, Algorithm, DecodingKey, EncodingKey, Header, Validation};
use once_cell::sync::Lazy;
use rand::{distributions::Alphanumeric, Rng};
use serde::{Deserialize, Serialize};
use thiserror::Error;
use tracing::{debug, error, info, instrument, trace, warn};

/// Time-to-live for an issued token (in minutes).
const SESSION_TTL_MINUTES: i64 = 60;

/// Expose a single global signing secret for now. In prod we rotate via KMS.
static JWT_SECRET: Lazy<String> = Lazy::new(|| {
    std::env::var("CHROMATICCANVAS_JWT_SECRET").unwrap_or_else(|_| {
        warn!("Using autogenerated JWT secret – set CHROMATICCANVAS_JWT_SECRET for security.");
        rand::thread_rng()
            .sample_iter(&Alphanumeric)
            .take(64)
            .map(char::from)
            .collect()
    })
});

/// ==========================================================================
///  Models
/// ==========================================================================

#[derive(Debug, Serialize, Deserialize)]
pub struct Claims {
    /// Stable user identifier, usually a UUID v4.
    pub sub: String,
    /// Unix timestamp when the token was issued.
    pub iat: i64,
    /// Unix timestamp when the token expires.
    pub exp: i64,
    /// Permission buckets understood by frontend & backend alike.
    pub roles: Vec<String>,
    /// Server-side session identifier. Lets us revoke *individual* tokens.
    pub sid: String,
}

/// A validated, still-active session passed to request handlers.
///
/// ```ignore
/// async fn my_handler(session: Session) -> impl Responder {
///     debug!("User {} has roles {:?}", session.user_id, session.roles);
///     HttpResponse::Ok().finish()
/// }
/// ```
#[derive(Clone, Debug)]
pub struct Session {
    pub user_id: String,
    pub roles: Arc<[String]>,
    pub issued_at: DateTime<Utc>,
    pub expires_at: DateTime<Utc>,
    /// Internally-generated session ID linking this token in the store.
    sid: String,
}

/// Server-side registry of active sessions.
#[derive(Debug, Default)]
pub struct SessionStore {
    // sid → (user_id, expires_at)
    inner: DashMap<String, (String, DateTime<Utc>)>,
}

impl SessionStore {
    pub fn new() -> Self {
        Self::default()
    }

    /// Put a fresh session into the store.
    fn insert(&self, claims: &Claims) {
        self.inner.insert(
            claims.sid.clone(),
            (claims.sub.clone(), Utc.timestamp_opt(claims.exp, 0).unwrap()),
        );
    }

    /// Check that `sid` is present and not expired.
    fn contains_active(&self, sid: &str) -> bool {
        self.inner
            .get(sid)
            .map(|v| v.1 > Utc::now())
            .unwrap_or(false)
    }

    /// Delete a session so remaining JWTs using it instantly fail.
    pub fn invalidate(&self, sid: &str) {
        self.inner.remove(sid);
    }

    /// Periodic house-keeping: clears every session past its `exp`.
    pub fn vacuum_expired(&self) {
        let now = Utc::now();
        self.inner
            .retain(|_, (_, exp)| *exp > now);
    }
}

/// Custom error variants for every failure stage.
#[derive(Debug, Error)]
pub enum SessionError {
    #[error("missing authorization header")]
    MissingHeader,
    #[error("unsupported authorization scheme")]
    MalformedHeader,
    #[error("invalid or expired token")]
    InvalidToken,
    #[error("session has been revoked")]
    Revoked,
    #[error("internal session store error")]
    Store(String),
}

impl ResponseError for SessionError {
    fn status_code(&self) -> StatusCode {
        match self {
            SessionError::MissingHeader | SessionError::MalformedHeader => StatusCode::BAD_REQUEST,
            SessionError::InvalidToken | SessionError::Revoked => StatusCode::UNAUTHORIZED,
            SessionError::Store(_) => StatusCode::INTERNAL_SERVER_ERROR,
        }
    }

    fn error_response(&self) -> actix_web::HttpResponse {
        use actix_web::HttpResponse;
        HttpResponse::build(self.status_code()).json(serde_json::json!({
            "error": self.to_string()
        }))
    }
}

/// ==========================================================================
///  Session Manager
/// ==========================================================================

#[derive(Clone)]
pub struct SessionManager {
    store: Arc<SessionStore>,
    encoding_key: EncodingKey,
    decoding_key: DecodingKey<'static>,
}

impl SessionManager {
    pub fn new(store: Arc<SessionStore>) -> Self {
        let secret = JWT_SECRET.as_bytes();
        Self {
            store,
            encoding_key: EncodingKey::from_secret(secret),
            decoding_key: DecodingKey::from_secret(secret),
        }
    }

    /// Create & register a new session, returning the raw JWT string.
    #[instrument(skip(self))]
    pub fn issue_token<S: Into<String>>(
        &self,
        user_id: S,
        roles: Vec<String>,
    ) -> Result<String, SessionError> {
        let user_id = user_id.into();
        let issued_at = Utc::now();
        let expires_at = issued_at + Duration::minutes(SESSION_TTL_MINUTES);

        // cryptographically-secure random session-id
        let sid: String = rand::thread_rng()
            .sample_iter(&Alphanumeric)
            .take(32)
            .map(char::from)
            .collect();

        let claims = Claims {
            sub: user_id.clone(),
            iat: issued_at.timestamp(),
            exp: expires_at.timestamp(),
            roles,
            sid: sid.clone(),
        };

        // Remember on server side for revocation support.
        self.store.insert(&claims);

        encode(&Header::new(Algorithm::HS256), &claims, &self.encoding_key)
            .map_err(|e| {
                error!("JWT encode failed: {e}");
                SessionError::Store("encode failure".into())
            })
    }

    /// Validate a token string and return a `Session`.
    #[instrument(skip(self, token))]
    pub fn validate_token(&self, token: &str) -> Result<Session, SessionError> {
        let token_data = decode::<Claims>(
            token,
            &self.decoding_key,
            &Validation::new(Algorithm::HS256),
        )
        .map_err(|e| {
            warn!("Failed JWT decode: {e}");
            SessionError::InvalidToken
        })?;

        let claims = token_data.claims;
        if !self.store.contains_active(&claims.sid) {
            return Err(SessionError::Revoked);
        }

        Ok(Session {
            user_id: claims.sub,
            roles: claims.roles.into(),
            issued_at: Utc.timestamp_opt(claims.iat, 0).unwrap(),
            expires_at: Utc.timestamp_opt(claims.exp, 0).unwrap(),
            sid: claims.sid,
        })
    }
}

/// ==========================================================================
///  Actix Extractor
/// ==========================================================================

impl FromRequest for Session {
    type Error = SessionError;
    type Future = Ready<Result<Self, Self::Error>>;

    fn from_request(req: &HttpRequest, _: &mut actix_web::dev::Payload) -> Self::Future {
        let manager = req
            .app_data::<Arc<SessionManager>>()
            .cloned()
            .expect("SessionManager not attached to app data");

        let auth_header = req
            .headers()
            .get(header::AUTHORIZATION)
            .and_then(|h| h.to_str().ok().map(String::from));

        ready(match auth_header {
            Some(header_value) => {
                // Expected format: `Bearer <token>`
                let mut parts = header_value.splitn(2, ' ');
                match (parts.next(), parts.next()) {
                    (Some(scheme), Some(token)) if scheme.eq_ignore_ascii_case("Bearer") => {
                        manager.validate_token(token)
                    }
                    _ => Err(SessionError::MalformedHeader),
                }
            }
            None => Err(SessionError::MissingHeader),
        })
    }
}

/// ==========================================================================
///  Revocation middleware
/// ==========================================================================
///
/// Combines automatic expiry vacuuming *and* a high-level `logout` helper
/// that can be injected into request extensions.

#[derive(Clone)]
pub struct SessionLayer {
    manager: Arc<SessionManager>,
}

impl SessionLayer {
    pub fn new(manager: Arc<SessionManager>) -> Self {
        Self { manager }
    }
}

impl<S, B> Transform<S, ServiceRequest> for SessionLayer
where
    S: Service<ServiceRequest, Response = ServiceResponse<B>, Error = actix_web::Error>
        + 'static,
    S::Future: 'static,
{
    type Response = ServiceResponse<B>;
    type Error = actix_web::Error;
    type InitError = ();
    type Transform = SessionLayerMiddleware<S>;
    type Future = Ready<Result<Self::Transform, Self::InitError>>;

    fn new_transform(&self, service: S) -> Self::Future {
        ready(Ok(SessionLayerMiddleware {
            service,
            manager: self.manager.clone(),
            last_vacuum: std::time::Instant::now(),
        }))
    }
}

pub struct SessionLayerMiddleware<S> {
    service: S,
    manager: Arc<SessionManager>,
    last_vacuum: std::time::Instant,
}

impl<S, B> Service<ServiceRequest> for SessionLayerMiddleware<S>
where
    S: Service<ServiceRequest, Response = ServiceResponse<B>, Error = actix_web::Error>
        + 'static,
    S::Future: 'static,
{
    type Response = ServiceResponse<B>;
    type Error = actix_web::Error;
    type Future = LocalBoxFuture<'static, Result<Self::Response, Self::Error>>;

    forward_ready!(service);

    fn call(&self, mut req: ServiceRequest) -> Self::Future {
        // Attach a handle that can be used from inside handlers:
        // `req.extensions_mut().get::<LogoutHandle>().unwrap()(sid);`
        let logout_handle = {
            let manager = self.manager.clone();
            Arc::new(move |sid: String| {
                manager.store.invalidate(&sid);
            }) as Arc<dyn Fn(String) + Send + Sync>
        };
        req.extensions_mut().insert(LogoutHandle(logout_handle));

        // Periodically run garbage collector
        if self.last_vacuum.elapsed() > StdDuration::from_secs(60) {
            self.manager.store.vacuum_expired();
        }

        let fut = self.service.call(req);
        Box::pin(async move { fut.await })
    }
}

/// Wrapper type because we cannot directly insert a `dyn Fn`.
#[derive(Clone)]
pub struct LogoutHandle(pub Arc<dyn Fn(String) + Send + Sync>);

/// Helper trait for ergonomics inside handlers.
pub trait LogoutExt {
    fn logout(&self, session: &Session);
}

impl LogoutExt for actix_web::HttpRequest {
    fn logout(&self, session: &Session) {
        if let Some(handle) = self.extensions().get::<LogoutHandle>() {
            (handle.0)(session.sid.clone());
        } else {
            warn!("LogoutHandle missing from request extensions");
        }
    }
}

#[cfg(test)]
mod tests {
    use super::*;
    use actix_web::{test, web, App};

    #[actix_web::test]
    async fn test_issue_and_validate() {
        let store = Arc::new(SessionStore::new());
        let manager = Arc::new(SessionManager::new(store));

        let token = manager
            .issue_token("user-123", vec!["artist".into()])
            .unwrap();

        let session = manager.validate_token(&token).unwrap();
        assert_eq!(session.user_id, "user-123");
        assert!(session.roles.contains(&"artist".into()));
    }

    #[actix_web::test]
    async fn extractor_integration() {
        let store = Arc::new(SessionStore::new());
        let manager = Arc::new(SessionManager::new(store.clone()));
        let token = manager
            .issue_token("abc", vec![])
            .expect("token issued");

        let app = test::init_service(
            App::new()
                .app_data(manager.clone())
                .wrap(SessionLayer::new(manager.clone()))
                .route(
                    "/",
                    web::get().to(|session: Session| async move {
                        format!("hello {}", session.user_id)
                    }),
                ),
        )
        .await;

        let req = test::TestRequest::with_uri("/")
            .insert_header((header::AUTHORIZATION, format!("Bearer {token}")))
            .to_request();

        let resp = test::call_and_read_body(&app, req).await;
        assert_eq!(resp, &b"hello abc"[..]);
    }

    #[actix_web::test]
    async fn revoked_token_fails() {
        let store = Arc::new(SessionStore::new());
        let manager = Arc::new(SessionManager::new(store.clone()));
        let token = manager
            .issue_token("abc", vec![])
            .expect("token issued");
        let claims = manager.validate_token(&token).unwrap();
        // Revoke it
        store.invalidate(&claims.sid);

        let result = manager.validate_token(&token);
        assert!(matches!(result, Err(SessionError::Revoked)));
    }
}
