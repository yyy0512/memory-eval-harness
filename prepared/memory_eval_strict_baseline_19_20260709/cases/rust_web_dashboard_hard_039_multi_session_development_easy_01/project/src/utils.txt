//! src/utils.rs
//!
//! A grab-bag of high–level helper functions and types that are used across
//! the entire ChromaticCanvas code-base.  The goal of this module is to keep
//! small but important cross-cutting concerns (configuration, cryptography,
//! time helpers, tracing helpers, etc.) in one well-documented place so that
//! the more domain-specific modules (REST handlers, MVC controllers, Yew
//! components, …) can stay lean and focused.
//!
//! Every public function has been designed with the following principles in
//! mind:
//!   1. Secure-by-default (e.g. Argon2id for password hashing, OS RNG).
//!   2. Reasonable ergonomics (clear error messages, sensible defaults).
//!   3. Async-friendly where appropriate (e.g. `spawn_blocking_traced`).
//!
//! NOTE: This file is compiled as `utils.rs` even though the prompt references
//!       `utils.txt`.  Rename as needed in your build script or `mod` tree.

#![allow(dead_code)]

use std::{
    fmt,
    fs,
    path::Path,
    time::{Duration, SystemTime, UNIX_EPOCH},
};

use once_cell::sync::Lazy;
use rand::{rngs::OsRng, RngCore};
use serde::Deserialize;
use thiserror::Error;
use tokio::task;
use tracing::{instrument, Span};

/// Convenient alias used throughout the project.
pub type Result<T, E = UtilError> = std::result::Result<T, E>;

/// Top-level error enumeration for utilities.
///
/// Domain-specific layers (e.g. controllers, repositories) are free to wrap or
/// convert these errors into their own types, but having one place where the
/// most common failure modes are described greatly improves debuggability.
///
/// All variants intentionally implement `std::error::Error` via `thiserror`.
#[derive(Error, Debug)]
pub enum UtilError {
    #[error("Configuration error: {0}")]
    Config(String),

    #[error("Cryptography error: {0}")]
    Crypto(String),

    #[error("I/O error: {0}")]
    Io(#[from] std::io::Error),

    #[error("Time error: {0}")]
    Time(String),

    #[error("Parsing error: {0}")]
    Parse(String),

    #[error("Unknown error: {0}")]
    Other(String),
}

/// ------------------------------------------------------------------------------------------------
/// Configuration helpers
/// ------------------------------------------------------------------------------------------------

/// Structure of the global application configuration.
///
/// The fields should mirror those defined in `Settings.toml` and/or environment
/// variables.  Only the values that _must_ be accessed globally live here;
/// component-specific options should reside closer to the component itself.
#[derive(Debug, Clone, Deserialize)]
pub struct AppConfig {
    /// Database connection URL (e.g. `postgres://user:pass@localhost/chromatic`)
    pub database_url: String,

    /// Secret used to sign JWT session tokens.
    pub jwt_secret: String,

    /// Listening address for the HTTP/HTTPS server.
    #[serde(default = "default_http_addr")]
    pub http_addr: String,

    /// Fully qualified path to the TLS certificate to enable SSL.
    #[serde(default)]
    pub tls_cert_path: Option<String>,

    /// Fully qualified path to the matching TLS private key.
    #[serde(default)]
    pub tls_key_path: Option<String>,

    /// If true, verbose tracing instrumentation will be enabled.
    #[serde(default)]
    pub debug_tracing: bool,
}

fn default_http_addr() -> String {
    "0.0.0.0:8080".into()
}

/// Lazily-initialised singleton configuration instance.
///
/// The first time the singleton is accessed we read settings off disk,
/// merge with environment overrides, and cache the result.
pub static CONFIG: Lazy<AppConfig> = Lazy::new(|| {
    load_config().unwrap_or_else(|err| {
        // Fail hard & early‽  Yes.  Configuration errors are non-recoverable.
        // Printing here ensures something shows in CI logs before panicking.
        eprintln!("FATAL — failed to load configuration: {err}");
        std::process::exit(1);
    })
});

/// Load the configuration from `Settings.toml`, `Settings.{env}.toml`, and
/// environment variables.  This mirrors what the `config` crate does but we
/// keep it minimal to avoid a public dependency explosion.
fn load_config() -> Result<AppConfig> {
    // 1. Attempt to read `Settings.toml`
    let mut builder = config::Config::builder();

    if Path::new("Settings.toml").exists() {
        builder = builder.add_source(config::File::with_name("Settings"));
    }

    // 2. Optional environment-specific file: `Settings.${RUN_MODE}.toml`
    if let Ok(run_mode) = std::env::var("RUN_MODE") {
        let file_name = format!("Settings.{run_mode}");
        if Path::new(&format!("{file_name}.toml")).exists() {
            builder = builder.add_source(config::File::with_name(&file_name));
        }
    }

    // 3. Environment variables take precedence (`CHROMA__DATABASE_URL`, etc.)
    builder = builder.add_source(
        config::Environment::with_prefix("CHROMA")
            .separator("__") // double-underscore => nested fields
            .try_parsing(true),
    );

    builder
        .build()
        .map_err(|e| UtilError::Config(e.to_string()))?
        .try_deserialize::<AppConfig>()
        .map_err(|e| UtilError::Config(e.to_string()))
}

/// ------------------------------------------------------------------------------------------------
/// Cryptographic helpers
/// ------------------------------------------------------------------------------------------------

/// Generate a cryptographically-secure random token of exactly `len` bytes,
/// returned as a URL-safe base64 string _without_ padding.
///
/// Useful for CSRF tokens, session identifiers, password resets, etc.
pub fn generate_secure_token(len: usize) -> Result<String> {
    if len == 0 {
        return Err(UtilError::Crypto("token length cannot be zero".into()));
    }

    let mut buf = vec![0u8; len];
    OsRng.fill_bytes(&mut buf);
    Ok(base64::engine::general_purpose::URL_SAFE_NO_PAD.encode(buf))
}

/// Hash a password using Argon2id.
///
/// Configuration deliberately errs on the side of strong defaults.  Tune
/// `mem_cost`/`time_cost` if you run on extremely constrained hardware.
pub fn hash_password(password: &str, salt: &[u8]) -> Result<String> {
    let config = argon2::Config {
        variant: argon2::Variant::Argon2id,
        version: argon2::Version::Version13,
        mem_cost: 64 * 1024, // 64 MiB
        time_cost: 3,
        lanes: 4,
        ..Default::default()
    };

    argon2::hash_encoded(password.as_bytes(), salt, &config)
        .map_err(|e| UtilError::Crypto(format!("argon2: {e}")))
}

/// Verify a password against an Argon2id hash.
///
/// Returns `Ok(true)` if the password is valid, `Ok(false)` otherwise.
pub fn verify_password(hash: &str, candidate_password: &str) -> Result<bool> {
    argon2::verify_encoded(hash, candidate_password.as_bytes())
        .map_err(|e| UtilError::Crypto(format!("argon2: {e}")))
}

/// Return the current UNIX timestamp in milliseconds since epoch.
#[inline]
pub fn now_millis() -> u128 {
    SystemTime::now()
        .duration_since(UNIX_EPOCH)
        .expect("system time before UNIX_EPOCH")
        .as_millis()
}

/// Format a `Duration` into a human-friendly relative string.
///
/// Examples:
///   * "just now"             (<   5 s)
///   * "12 seconds ago"       (<  60 s)
///   * "7 minutes ago"        (<  60 m)
///   * "3 hours ago"          (<  24 h)
///   * "2 days ago"
///
/// This is used in the UI to display timestamps without sending the full
/// chrono dependency tree down to the browser via WASM.
pub fn relative_time(d: Duration) -> String {
    match d.as_secs() {
        0..=4 => "just now".to_string(),
        5..=59 => format!("{} seconds ago", d.as_secs()),
        60..=3599 => format!("{} minutes ago", d.as_secs() / 60),
        3600..=86_399 => format!("{} hours ago", d.as_secs() / 3600),
        _ => format!("{} days ago", d.as_secs() / 86_400),
    }
}

/// Spawn a blocking task and propagate the current tracing [`Span`].
///
/// This is a small but extremely handy wrapper around `tokio::task::spawn_blocking`
/// that maintains the span context so that `tracing` logs inside the blocking
/// task still show up correctly nested alongside async work.
///
/// Example:
/// ```no_run
/// use utils::spawn_blocking_traced;
///
/// let result = spawn_blocking_traced(|| std::fs::read_to_string("large_file.txt"))
///     .await
///     .expect("blocking task failed");
/// ```
#[instrument(level = "trace", skip(f))]
pub async fn spawn_blocking_traced<F, R>(f: F) -> Result<R, task::JoinError>
where
    F: FnOnce() -> R + Send + 'static,
    R: Send + 'static,
{
    let span = Span::current();
    task::spawn_blocking(move || {
        let _entered = span.enter();
        f()
    })
    .await
}

/// Convert a hex triplet (`#RRGGBB`) to an `(r, g, b)` tuple.
pub fn parse_hex_color(hex: &str) -> Result<(u8, u8, u8)> {
    let hex = hex.strip_prefix('#').unwrap_or(hex);
    if hex.len() != 6 || !hex.chars().all(|c| c.is_ascii_hexdigit()) {
        return Err(UtilError::Parse(format!(
            "invalid hex colour literal: {hex}"
        )));
    }

    let r = u8::from_str_radix(&hex[0..2], 16).unwrap();
    let g = u8::from_str_radix(&hex[2..4], 16).unwrap();
    let b = u8::from_str_radix(&hex[4..6], 16).unwrap();
    Ok((r, g, b))
}

/// Linearly blend two colours (`ratio` ∈ 0.0‥1.0).
pub fn blend_colors(a: (u8, u8, u8), b: (u8, u8, u8), ratio: f32) -> (u8, u8, u8) {
    let clamp = |v: f32| v.max(0.0).min(255.0) as u8;

    let r = clamp(a.0 as f32 * (1.0 - ratio) + b.0 as f32 * ratio);
    let g = clamp(a.1 as f32 * (1.0 - ratio) + b.1 as f32 * ratio);
    let b = clamp(a.2 as f32 * (1.0 - ratio) + b.2 as f32 * ratio);
    (r, g, b)
}

/// Atomically write data to a file, replacing any previous content.
///
/// The data is first written to `<path>.tmp` and then persisted via a
/// rename-operation (atomic on most POSIX & NTFS filesystems).
pub fn atomic_write<P: AsRef<Path>>(path: P, data: &[u8]) -> Result<()> {
    let tmp_path = path
        .as_ref()
        .with_extension("tmp");

    fs::write(&tmp_path, data)?;
    fs::rename(&tmp_path, &path)?;
    Ok(())
}

/// A small helper used throughout debug logs to shorten large collections.
pub struct Preview<'a, T: fmt::Debug>(&'a [T], usize);

impl<'a, T: fmt::Debug> Preview<'a, T> {
    /// Create a new preview that limits the maximum number of items shown.
    pub fn new(slice: &'a [T], max_items: usize) -> Self {
        Self(slice, max_items)
    }
}

impl<T: fmt::Debug> fmt::Display for Preview<'_, T> {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        let preview_len = self.0.len().min(self.1);
        write!(f, "[")?;
        for (idx, item) in self.0.iter().take(preview_len).enumerate() {
            if idx > 0 {
                write!(f, ", ")?;
            }
            write!(f, "{item:?}")?;
        }
        if self.0.len() > self.1 {
            write!(f, ", … ({} more)", self.0.len() - self.1)?;
        }
        write!(f, "]")
    }
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn test_hex_parser() {
        assert_eq!(parse_hex_color("#ffffff").unwrap(), (0xff, 0xff, 0xff));

        let red = (255, 0, 0);
        let blue = (0, 0, 255);
        assert_eq!(blend_colors(red, blue, 0.5), (127, 0, 127));
    }

    #[tokio::test]
    async fn test_spawn_blocking_traced() {
        let res = spawn_blocking_traced(|| 1 + 1).await.unwrap();
        assert_eq!(res, 2);
    }
}
